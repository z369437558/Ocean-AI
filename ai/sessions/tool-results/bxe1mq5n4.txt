   485	/**
   486	 * A full page load ends the store's life while a request may still be on its way. What the next
   487	 * page needs to carry on safely - the course and above all the key - is kept in the tab's storage.
   488	 */
   489	describe('bulkIssueStore: attempts without a known outcome survive a page load', () => {
   490	  /** Stands in for window.sessionStorage: it outlives the store, like the tab outlives the page. */
   491	  function tabStorage(initial = {}) {
   492	    const items = new Map(Object.entries(initial));
   493	    return {
   494	      items,
   495	      getItem: (name) => (items.has(name) ? items.get(name) : null),
   496	      setItem: (name, value) => items.set(name, String(value)),
   497	      removeItem: (name) => items.delete(name),
   498	    };
   499	  }
   500	
   501	  it('brings a request that was in flight back as an unknown outcome with the same course and key', () => {
   502	    const storage = tabStorage();
   503	    createBulkIssueStore(storage).submit(pendingApi(), attempt); // never answered: the page is reloaded first
   504	
   505	    const afterReload = createBulkIssueStore(storage);
   506	    const { run, settled } = afterReload.read(ALICE);
   507	    expect(run.status).toBe('failed');
   508	    expect(run.attempt).toEqual(attempt);
   509	    expect(run.error).toBeInstanceOf(ApiError);
   510	    expect(run.error.message).toBe('The page was reloaded or left before the answer from the server arrived.');
   511	    expect(writeOutcome(run.error)).toBe('unknown'); // "may or may not have gone through", with "Retry safely"
   512	    expect(settled).toBe(0);
   513	    expect(afterReload.read(BOB)).toEqual({ run: { status: 'idle' }, settled: 0 });
   514	
   515	    // the retry from the new page re-sends the same key
   516	    const api = pendingApi();
   517	    afterReload.submit(api, run.attempt);
   518	    expect(api.bulkIssue).toHaveBeenCalledWith(1, 'key-1');
   519	  });

   523	    ['an HTTP 500', new ApiError(500, 'internal_error', 'Unexpected server error'), 'unknown'],
   524	    ['a busy database', new ApiError(503, 'database_busy', 'The database is busy; retry the request shortly'), 'not_run'],
   525	    ['an unexpected exception', new TypeError('x is undefined'), 'unknown'],
   526	  ])('brings the retry prompt after %s back as it was', async (_case, error, outcome) => {
