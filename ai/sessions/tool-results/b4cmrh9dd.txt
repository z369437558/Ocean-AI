    13	  return `${tenantCode}-${String(sequence).padStart(7, '0')}`;

    16	/** Reserves `count` consecutive serial numbers for a tenant and returns the first sequence value. */
    17	export function reserveSerials(db, tenantId, count = 1) {
    18	  const { changes } = db.run('UPDATE tenants SET last_serial = last_serial + @count WHERE id = @tenantId', {
    19	    count,
    20	    tenantId,
    21	  });
    22	  if (changes !== 1) throw new Error(`Unknown tenant ${tenantId}`);
    23	  const { last_serial: last } = db.get('SELECT last_serial FROM tenants WHERE id = @tenantId', { tenantId });
    24	  return last - count + 1;
    25	}

    44	/**
    45	 * Issues one certificate and writes its `certificate.issued` audit event.
    46	 *
    47	 * `tenant` is `{ id, code, timezone }`, `actor` is `{ adminId, email }`.
    48	 * Throws if the learner already has a certificate for the course (UNIQUE constraint).
    49	 */
    50	export function issueCertificate(db, { tenant, actor, course, learner, enrollmentId, issuedAt, bulkOperationId = null, auditDetails = {} }) {
    51	  const issuedAtIso = toUtcIso(issuedAt);
    52	  const serialNumber = formatSerial(tenant.code, reserveSerials(db, tenant.id, 1));
    53	  const certificateId = db.run(
    54	    `INSERT INTO certificates (tenant_id, course_id, learner_id, enrollment_id, serial_number, learner_name, course_title,
    55	                               status, issued_at, issued_local_date, issued_by_admin_id, bulk_operation_id)
    56	     VALUES (@tenantId, @courseId, @learnerId, @enrollmentId, @serialNumber, @learnerName, @courseTitle,
    57	             'active', @issuedAt, @issuedLocalDate, @issuedByAdminId, @bulkOperationId)`,
    58	    {
    59	      tenantId: tenant.id,
    60	      courseId: course.id,
    61	      learnerId: learner.id,
    62	      enrollmentId: enrollmentId ?? null,
    63	      serialNumber,
    64	      learnerName: learner.name,
    65	      courseTitle: course.title,
    66	      issuedAt: issuedAtIso,
    67	      issuedLocalDate: localDate(issuedAtIso, tenant.timezone),
    68	      issuedByAdminId: actor.adminId ?? null,
    69	      bulkOperationId,
    70	    },
    71	  ).lastInsertRowid;
    72	
    73	  recordAuditEvent(db, {
    74	    tenantId: tenant.id,
    75	    actorAdminId: actor.adminId,
    76	    actorEmail: actor.email,
    77	    action: AUDIT_ACTIONS.ISSUED,
    78	    certificateId,
    79	    bulkOperationId,
    80	    createdAt: issuedAtIso,
    81	    details: {
    82	      serialNumber,
    83	      learnerId: learner.id,
    84	      learnerName: learner.name,
    85	      courseId: course.id,
    86	      courseTitle: course.title,
    87	      ...auditDetails,
    88	    },
    89	  });
    90	
    91	  return { id: certificateId, serialNumber, issuedAt: issuedAtIso };
    92	}

    94	/**
    95	 * Revokes an active certificate and writes its `certificate.revoked` audit event.
    96	 * Returns false (and writes nothing) when the certificate is not active any more, which is what
    97	 * makes repeated or racing revoke requests harmless.
    98	 */
    99	export function revokeCertificate(db, { tenantId, certificateId, actor, revokedAt, reason = null }) {
   100	  const revokedAtIso = toUtcIso(revokedAt);
   101	  const { changes } = db.run(
   102	    `UPDATE certificates
   103	        SET status = 'revoked', revoked_at = @revokedAt, revoked_by_admin_id = @adminId, revocation_reason = @reason
   104	      WHERE id = @certificateId AND tenant_id = @tenantId AND status = 'active'`,
   105	    { revokedAt: revokedAtIso, adminId: actor.adminId ?? null, reason, certificateId, tenantId },
   106	  );
   107	  if (changes !== 1) return false;
   108	
   109	  const { serial_number: serialNumber } = db.get('SELECT serial_number FROM certificates WHERE id = @certificateId', {
   110	    certificateId,
   111	  });
   112	  recordAuditEvent(db, {
   113	    tenantId,
   114	    actorAdminId: actor.adminId,
   115	    actorEmail: actor.email,
   116	    action: AUDIT_ACTIONS.REVOKED,
   117	    certificateId,
   118	    createdAt: revokedAtIso,
   119	    details: { serialNumber, reason, previousStatus: 'active' },
   120	  });
   121	  return true;
   122	}

   131	export const SKIP_REASONS = Object.freeze({
   132	  ALREADY_CERTIFIED: 'already_certified',
   133	  CERTIFICATE_REVOKED: 'certificate_revoked',
   134	  NOT_COMPLETED: 'not_completed',

   137	const ENROLLMENT_LABELS = { not_started: 'not started', in_progress: 'in progress', completed: 'completed' };

   139	/**
   140	 * Decides what a bulk issue does with one roster row.
   141	 * Eligible = enrollment completed and no certificate has ever been issued for (course, learner).
   142	 */
   143	export function classifyEnrollment(row) {
   144	  if (row.certificate_id !== null && row.certificate_status === 'revoked') {
   145	    return {
   146	      eligible: false,
   147	      reason: SKIP_REASONS.CERTIFICATE_REVOKED,
   148	      message: `Certificate ${row.serial_number} was revoked; revoked certificates are not re-issued automatically`,
   149	    };
   150	  }
   151	  if (row.certificate_id !== null) {
   152	    return {
   153	      eligible: false,
   154	      reason: SKIP_REASONS.ALREADY_CERTIFIED,
   155	      message: `Already certified (certificate ${row.serial_number})`,
   156	    };
   157	  }
   158	  if (row.enrollment_status !== 'completed') {
   159	    return {
   160	      eligible: false,
   161	      reason: SKIP_REASONS.NOT_COMPLETED,
   162	      message: `Course not completed (enrollment is ${ENROLLMENT_LABELS[row.enrollment_status] ?? row.enrollment_status})`,
   163	    };
   164	  }
   165	  return { eligible: true, reason: null, message: 'Completed, not yet certified' };
   166	}

   168	const ROSTER_SQL = `
   169	  SELECT e.id AS enrollment_id, e.status AS enrollment_status, e.completed_at,
   170	         l.id AS learner_id, l.full_name AS learner_name,
   171	         c.id AS certificate_id, c.serial_number, c.status AS certificate_status
   172	    FROM enrollments e
   173	    JOIN learners l ON l.id = e.learner_id AND l.tenant_id = e.tenant_id
   174	    LEFT JOIN certificates c
   175	           ON c.tenant_id = e.tenant_id AND c.course_id = e.course_id AND c.learner_id = e.learner_id
   176	   WHERE e.tenant_id = @tenantId AND e.course_id = @courseId
   177	   ORDER BY l.full_name COLLATE NOCASE, l.id`;

   204	/** Roster page plus per-course totals for the bulk-issue screen. */
   205	export function getCourseRoster(db, actor, courseId, query) {
   206	  const page = queryInteger(query, ['page'], { min: 1, max: 1_000_000, fallback: 1 });
   207	  const pageSize = queryInteger(query, ['pageSize', 'page_size', 'per_page', 'limit'], { min: 1, max: 100, fallback: 50 });
   208	
   209	  return db.transaction(
   210	    () => {
   211	      const course = findCourse(db, actor, courseId);
   212	      if (!course) throw notFound('Course');
   213	
   214	      const params = { tenantId: actor.tenantId, courseId: course.id };
   215	      const totals = db.get(
   216	        `SELECT COUNT(*) AS enrolled,
   217	                COALESCE(SUM(e.status = 'completed'), 0) AS completed,
   218	                COALESCE(SUM(c.status = 'active'), 0) AS certified,
   219	                COALESCE(SUM(c.status = 'revoked'), 0) AS revoked,
   220	                COALESCE(SUM(e.status = 'completed' AND c.id IS NULL), 0) AS eligible
   221	           FROM enrollments e
   222	           LEFT JOIN certificates c
   223	                  ON c.tenant_id = e.tenant_id AND c.course_id = e.course_id AND c.learner_id = e.learner_id
   224	          WHERE e.tenant_id = @tenantId AND e.course_id = @courseId`,
   225	        params,
   226	      );
   227	      const rows = db.all(`${ROSTER_SQL} LIMIT @limit OFFSET @offset`, {
   228	        ...params,
   229	        limit: pageSize,
   230	        offset: (page - 1) * pageSize,
   231	      });
   232	
   233	      return {
   234	        course,
   235	        summary: {
   236	          enrolled: totals.enrolled,
   237	          completed: totals.completed,
   238	          incomplete: totals.enrolled - totals.completed,
   239	          certified: totals.certified,
   240	          revoked: totals.revoked,
   241	          eligible: totals.eligible,
   242	        },
   243	        items: rows.map((row) => {
   244	          const verdict = classifyEnrollment(row);
   245	          return {
   246	            learnerId: row.learner_id,
   247	            learnerName: row.learner_name,
   248	            enrollmentId: row.enrollment_id,
   249	            enrollmentStatus: row.enrollment_status,
   250	            completedAt: row.completed_at,
   251	            completedLocal: row.completed_at ? localDateTime(row.completed_at, actor.timezone) : null,
   252	            certificateId: row.certificate_id,
   253	            serialNumber: row.serial_number,
   254	            certificateStatus: row.certificate_status,
   255	            eligible: verdict.eligible,
   256	            reason: verdict.reason,
   257	            message: verdict.message,
   258	          };
   259	        }),
   260	        page,
   261	        pageSize,
   262	        total: totals.enrolled,
   263	        totalPages: Math.max(1, Math.ceil(totals.enrolled / pageSize)),
   264	      };
   265	    },
   266	    { immediate: false },
   267	  );
   268	}

   306	/** Express middleware: resolves `req.actor` or rejects the request with 401. */
   307	export function requireActor(db) {
   308	  return function actorContext(req, _res, next) {
   309	    const email = (req.get(ACTOR_HEADER) ?? '').trim();
   310	    if (!email) {
   311	      throw new HttpError(401, 'actor_required', `Missing ${ACTOR_HEADER} header`);
   312	    }
   313	    const actor = findActorByEmail(db, email);
   314	    if (!actor) {
   315	      throw new HttpError(401, 'unknown_actor', `No admin account matches ${ACTOR_HEADER} "${email.slice(0, 120)}"`);
   316	    }
   317	    req.actor = actor;
   318	    next();
   319	  };
   320	}

   323	const NEEDS_QUOTING = /[",\r\n]/;
   324	const FORMULA_TRIGGER = /^[=+\-@\t\r]/;

   326	export function csvCell(value) {
   327	  let text = value === null || value === undefined ? '' : String(value);
   328	  // A cell that starts with = + - @ is executed as a formula by Excel/Sheets. Learner names are
   329	  // user-controlled, so neutralise those with a leading apostrophe (OWASP CSV-injection guidance).
   330	  if (FORMULA_TRIGGER.test(text)) text = `'${text}`;
   331	  return NEEDS_QUOTING.test(text) ? `"${text.replaceAll('"', '""')}"` : text;
   332	}

   335	  let line = '';

   337	    if (i > 0) line += ',';
   338	    line += csvCell(cells[i]);
   339	  }
   340	  return `${line}\n`;
   341	}

   362	/**
   363	 * JSON with object keys sorted at every level, so logically equal payloads serialise identically.
   364	 * Iterative on purpose: request bodies are attacker-controlled and JSON.parse accepts nesting far
   365	 * deeper than the call stack, so a recursive walk could be made to overflow.
   366	 */
   367	export function canonicalJson(value) {
   368	  const out = [];
   369	  // Work stack of things still to emit: literal strings, or { value } entries to serialise.
   370	  const stack = [{ value }];
   371	  while (stack.length > 0) {
   372	    const item = stack.pop();
   373	    if (typeof item === 'string') {
   374	      out.push(item);
   375	      continue;
   376	    }
   377	    const current = item.value;
   378	    if (current === null || typeof current !== 'object') {
   379	      out.push(JSON.stringify(current) ?? 'null');
   380	    } else if (Array.isArray(current)) {
   381	      stack.push(']');
   382	      for (let i = current.length - 1; i >= 0; i -= 1) {
   383	        stack.push({ value: current[i] });
   384	        if (i > 0) stack.push(',');
   385	      }
   386	      out.push('[');
   387	    } else {
   388	      const keys = Object.keys(current)
   389	        .filter((key) => current[key] !== undefined)
   390	        .sort();
   391	      stack.push('}');
   392	      for (let i = keys.length - 1; i >= 0; i -= 1) {
   393	        stack.push({ value: current[keys[i]] });
   394	        stack.push(`${JSON.stringify(keys[i])}:`);
   395	        if (i > 0) stack.push(',');
   396	      }
   397	      out.push('{');
   398	    }
   399	  }
   400	  return out.join('');
   401	}

   403	/**
   404	 * Stable hash of a request, used to tell an honest retry (same key, same request) apart from
   405	 * idempotency-key misuse (same key, different request).
   406	 */
   407	export function requestFingerprint(method, route, body) {
   408	  return createHash('sha256')
   409	    .update(`${method.toUpperCase()} ${route}\n${canonicalJson(body ?? null)}`)
   410	    .digest('hex');
   411	}
   412	/** Small request-parameter parsers that fail with a 400 instead of guessing. */
   413	import { badRequest } from './errors.js';

   415	const CONTROL_CHARS = /[\u0000-\u001f\u007f]/;
   416	const CONTROL_CHARS_EXCEPT_LINE_BREAKS = /[\u0000-\u0008\u000b\u000c\u000e-\u001f\u007f]/;
   417	
   418	/**
   419	 * Rejects text that could not be stored and read back faithfully: NUL and other control
   420	 * characters, and unpaired surrogates (which are not valid UTF-8 once written to SQLite).
   421	 */
   422	export function assertPlainText(value, label, { allowLineBreaks = false } = {}) {
   423	  const forbidden = allowLineBreaks ? CONTROL_CHARS_EXCEPT_LINE_BREAKS : CONTROL_CHARS;
   424	  if (forbidden.test(value) || !value.isWellFormed()) {
   425	    throw badRequest(`${label} contains characters that are not allowed (control characters or malformed Unicode)`);
   426	  }
   427	}

   429	/** First of the given query parameter names that is present, as a trimmed string. */
   430	export function queryString(query, names, { maxLength = 200 } = {}) {
   431	  for (const name of names) {
   432	    const value = query[name];
   433	    if (value === undefined) continue;
   434	    if (typeof value !== 'string') throw badRequest(`Query parameter "${name}" must be a single value`);
   435	    const text = value.trim();
   436	    assertPlainText(text, `Query parameter "${name}"`);
   437	    if (text.length > maxLength) throw badRequest(`Query parameter "${name}" must be at most ${maxLength} characters`);
   438	    return text;
   439	  }
   440	  return undefined;
   441	}
   442	
   443	export function queryInteger(query, names, { min, max, fallback }) {
   444	  for (const name of names) {
   445	    if (query[name] === undefined) continue;
   446	    const text = queryString(query, [name]);
   447	    if (!/^\d{1,9}$/.test(text)) throw badRequest(`Query parameter "${name}" must be a whole number`);
   448	    const value = Number(text);
   449	    if (value < min || value > max) throw badRequest(`Query parameter "${name}" must be between ${min} and ${max}`);
   450	    return value;
   451	  }
   452	  return fallback;
   453	}

   459	/**
   460	 * Time-zone helpers built on the ICU data that ships with Node.js (no extra dependency).
   461	 *
   462	 * Certificates store their issue instant in UTC. Everything a tenant admin sees or sorts by is
   463	 * that instant converted to the tenant's IANA time zone, DST rules included.
   464	 */
   465	
   466	const formatters = new Map();
   467	
   468	function formatterFor(timeZone) {
   469	  let formatter = formatters.get(timeZone);
   470	  if (!formatter) {
   471	    formatter = new Intl.DateTimeFormat('en-CA', {
   472	      timeZone,
   473	      hourCycle: 'h23',
   474	      year: 'numeric',
   475	      month: '2-digit',
   476	      day: '2-digit',
   477	      hour: '2-digit',
   478	      minute: '2-digit',
   479	      second: '2-digit',
   480	    });
   481	    formatters.set(timeZone, formatter);
   482	  }
   483	  return formatter;
   484	}

   486	export function isValidTimeZone(timeZone) {
   487	  try {
   488	    formatterFor(timeZone);
   489	    return true;
   490	  } catch {
   491	    return false;
   492	  }
   493	}

   495	function toDate(instant) {
   496	  const date = instant instanceof Date ? instant : new Date(instant);
   497	  if (Number.isNaN(date.getTime())) throw new RangeError(`Invalid timestamp: ${String(instant)}`);
   498	  return date;
   499	}

   501	/** Wall-clock fields ({ year, month, day, hour, minute, second } as zero-padded strings). */
   502	export function zonedParts(instant, timeZone) {
   503	  const parts = {};
   504	  for (const part of formatterFor(timeZone).formatToParts(toDate(instant))) {
   505	    if (part.type !== 'literal') parts[part.type] = part.value;
   506	  }
   507	  return parts;
   508	}

   522	/** UTC offset in force at that instant, e.g. `+11:00` (Sydney in January) or `+10:00` (Sydney in July). */
   523	export function utcOffset(instant, timeZone) {
   524	  const date = toDate(instant);
   525	  const p = zonedParts(date, timeZone);
   526	  const wallClockAsUtc = Date.UTC(+p.year, +p.month - 1, +p.day, +p.hour, +p.minute, +p.second);
   527	  const wholeSeconds = Math.floor(date.getTime() / 1000) * 1000;
   528	  const minutes = Math.round((wallClockAsUtc - wholeSeconds) / 60_000);
   529	  const abs = Math.abs(minutes);
   530	  const pad = (n) => String(n).padStart(2, '0');
   531	  return `${minutes < 0 ? '-' : '+'}${pad(Math.floor(abs / 60))}:${pad(abs % 60)}`;
   532	}

   534	/**
   535	 * Returns a memoising `instant -> 'YYYY-MM-DD HH:mm'` function for hot loops (CSV export).
   536	 * Bulk operations stamp many certificates with the same instant, so the cache hits often.
   537	 */
   538	export function createLocalDateTimeFormatter(timeZone, { maxEntries = 50_000 } = {}) {
   539	  const cache = new Map();
   540	  return (instant) => {
   541	    let text = cache.get(instant);
   542	    if (text === undefined) {
   543	      text = localDateTime(instant, timeZone);
   544	      if (cache.size >= maxEntries) cache.clear();
   545	      cache.set(instant, text);
   546	    }
   547	    return text;
   548	  };
   549	}

   572	/**
   573	 * Runtime configuration, all optional: PORT, HOST, DB_PATH, STATIC_DIR, LOG_REQUESTS.
   574	 * (The database layer reads two more settings itself: SQLITE_DRIVER and DB_BUSY_TIMEOUT_MS.)

   576	export function loadConfig(env = process.env) {
   577	  const port = env.PORT === undefined || env.PORT === '' ? 3000 : Number(env.PORT);
   578	  if (!Number.isInteger(port) || port < 0 || port > 65535) {
   579	    throw new Error(`PORT must be an integer between 0 and 65535 (got "${env.PORT}")`);
   580	  }
   581	  return {
   582	    port,
   583	    host: env.HOST || undefined, // undefined = all interfaces
   584	    dbPath: env.DB_PATH ? resolveUserPath(env.DB_PATH, env) : DEFAULT_DB_PATH,
   585	    staticDir: env.STATIC_DIR ? resolveUserPath(env.STATIC_DIR, env) : DEFAULT_STATIC_DIR,
   586	    logRequests: env.LOG_REQUESTS !== '0',
   587	  };
