    14	/**
    15	 * Confirmation step for revoking: nothing is sent until the admin explicitly confirms.
    16	 *
    17	 * A failed request is reported for what it says about the server (see writeOutcome in api.js):
    18	 *
    19	 * - A definite "no" (4xx, busy database) means the certificate is as it was.
    20	 * - An unknown outcome leaves open whether the server carried the request out. `onOutcomeUnknown`
    21	 *   then makes the drawer ask the server, and the drawer remembers it as `outcomeOpen`. While that
    22	 *   is set the dialog no longer claims "was not revoked" for a later attempt either, however
    23	 *   definite its failure: the earlier request may have gone through, or may yet (the server can
    24	 *   keep a request waiting for the database longer than a gateway waits for the server).
    25	 * - `revokedOnServer` reports that the drawer found the certificate revoked. That settles the
    26	 *   matter, whatever happened to the requests: the question and the retry give way to the
    27	 *   confirmed statement and a plain "Close".
    28	 */
    29	function RevokeDialog({ api, certificate, outcomeOpen, revokedOnServer, onCancel, onRevoked, onOutcomeUnknown }) {
    30	  const [reason, setReason] = useState('');
    31	  // outcome: what the failed attempt means - 'unknown' | 'rejected' | 'not_run' (writeOutcome).
    32	  const [state, setState] = useState({ status: 'idle', error: null, outcome: null });
    33	  const submitting = useRef(false);
    34	  const bodyRef = useRef(null);
    35	  const messageRef = useRef(null);
    36	  const confirmRef = useRef(null);
    37	  const closeRef = useRef(null);
    38	
    39	  // Revoking cannot be undone, so "revoked" stays true once the server has said it - also if a
    40	  // later reload of the drawer fails and leaves it without a certificate to show.
    41	  const [knownRevoked, setKnownRevoked] = useState(false);
    42	  if (revokedOnServer && !knownRevoked) setKnownRevoked(true);
    43	
    44	  const busy = state.status === 'submitting';
    45	  // While a retry is on its way the form stays as it is; its answer decides what is shown next.
    46	  const settled = knownRevoked && !busy;
    47	
    48	  async function confirm() {
    49	    if (submitting.current) return; // ignore double clicks while the request is on its way
    50	    submitting.current = true;
    51	    setState({ status: 'submitting', error: null, outcome: null });
    52	    let updated;
    53	    try {
    54	      updated = await api.revokeCertificate(certificate.id, reason.trim() || undefined);
    55	    } catch (error) {
    56	      submitting.current = false;
    57	      // Only a 4xx or a busy database (503, nothing was written) is a definite "no".
    58	      const outcome = writeOutcome(error);
    59	      setState({ status: 'error', error, outcome });
    60	      // Ask the server - also after a definite "no" while an earlier outcome is still open: the
    61	      // reload may settle that one meanwhile.
    62	      if (outcome === 'unknown' || outcomeOpen) onOutcomeUnknown();
    63	      return;
    64	    }
    65	    onRevoked(updated);
    66	  }
    67	
    68	  useEffect(() => {
    69	    if (state.status !== 'error' && !settled) return;
    70	    if (settled) {
    71	      // The form, and with it whatever had the focus, has just been replaced.
    72	      closeRef.current?.focus();
    73	    } else if (!bodyRef.current?.closest('[role="dialog"]')?.contains(document.activeElement)) {
    74	      // The buttons were disabled while the request was on its way, and a browser takes the focus
    75	      // away from a disabled button (to <body>, i.e. behind this dialog): bring it back.
    76	      confirmRef.current?.focus();
    77	    }
    78	    // On a short screen the dialog scrolls inside; what was just reported must not stay below the fold.
    79	    messageRef.current?.scrollIntoView?.({ block: 'nearest' });
    80	  }, [state, settled]);
    81	
    82	  return (
    83	    <Dialog onClose={onCancel} locked={busy} labelledBy="revoke-title" describedBy={settled ? 'revoke-confirmed' : undefined}>
    84	      {settled ? (
    85	        <div className="dialog-body" ref={bodyRef} data-testid="revoke-dialog">
    86	          <h2 id="revoke-title">Certificate revoked</h2>
    87	          <div className="dialog-scroll">
    88	            <p>
    89	              <span className="mono">{certificate.serialNumber}</span> was issued to <strong>{certificate.learnerName}</strong> for{' '}
    90	              <strong>{certificate.courseTitle}</strong>.
    91	            </p>
    92	            <p className="note note-success" role="status" id="revoke-confirmed" ref={messageRef} data-testid="revoke-confirmed">
    93	              Checked with the server: this certificate is revoked now. Nothing more needs to be done.
    94	            </p>
    95	            {state.status === 'error' && state.outcome !== 'unknown' ? (
    96	              <p className="muted small">The last attempt was not carried out: {state.error.message}</p>
    97	            ) : null}
    98	          </div>
    99	          <div className="dialog-actions">
   100	            <button type="button" className="btn btn-primary" onClick={onCancel} ref={closeRef} data-testid="revoke-close">
   101	              Close
   102	            </button>
   103	          </div>
   104	        </div>
   105	      ) : (
   106	        <form
   107	          className="dialog-body"
   108	          ref={bodyRef}
   109	          onSubmit={(event) => {
   110	            event.preventDefault();
   111	            confirm();
   112	          }}
   113	          data-testid="revoke-dialog"
   114	        >
   115	          <h2 id="revoke-title">Revoke this certificate?</h2>
   116	          {/* Only this part scrolls when the screen is too short; title and buttons stay in reach. */}
   117	          <div className="dialog-scroll">
   118	            <p>
   119	              <span className="mono">{certificate.serialNumber}</span> was issued to <strong>{certificate.learnerName}</strong> for{' '}
   120	              <strong>{certificate.courseTitle}</strong>. After revocation it is no longer valid. This cannot be undone, and
   121	              the action is recorded in the audit trail.
   122	            </p>
   123	            <label className="field">
   124	              <span className="field-label">Reason (optional)</span>
   125	              <textarea
   126	                rows={3}
   127	                maxLength={MAX_REASON_LENGTH}
   128	                value={reason}
   129	                disabled={busy}
   130	                placeholder="e.g. Issued in error"
   131	                onChange={(event) => setReason(event.target.value)}
   132	                data-testid="revoke-reason"
   133	              />
   134	            </label>
   135	            {state.status === 'error' ? (
   136	              <p className="note note-error" role="alert" ref={messageRef}>
   137	                {state.outcome === 'unknown' ? (
   138	                  <>
   139	                    <strong>The certificate may or may not have been revoked.</strong>{' '}
   140	                    {state.error?.message || 'Unexpected error'}
   141	                    <br />
   142	                    Trying again is safe: a certificate that is already revoked is left unchanged.
   143	                  </>
   144	                ) : outcomeOpen ? (
   145	                  <>
   146	                    <strong>The certificate may or may not have been revoked.</strong> This attempt was not carried out:{' '}
   147	                    {state.error.message}
   148	                    <br />

   151	                  </>
   152	                ) : (
   153	                  <>The certificate was not revoked: {state.error.message}</>
   154	                )}
   155	              </p>
   156	            ) : null}
   157	          </div>
   158	          <div className="dialog-actions">
   159	            <button type="button" className="btn" onClick={onCancel} disabled={busy}>
   160	              Cancel
   161	            </button>
   162	            <button type="submit" className="btn btn-danger" disabled={busy} ref={confirmRef} data-testid="revoke-confirm">
   163	              {busy ? <Spinner label="Revoking…" /> : state.status === 'error' ? 'Try again' : 'Revoke certificate'}
   164	            </button>
   165	          </div>
   166	        </form>

   190	  const [unverified, setUnverified] = useState(null);
   191	  const onScreen = useRef(null);
   192	  onScreen.current = detail.data;
   193	  // A revoke request made from this drawer ended without a known outcome (see RevokeDialog).
   194	  const [outcomeOpen, setOutcomeOpen] = useState(false);
   195	  const certificate = revoked ?? detail.data;
   196	  const verifying = revoked === null && unverified !== null && detail.data === unverified;
   197	
   198	  // When the confirmation closes, its own clean-up hands the focus back to the element that opened
   199	  // it - if that very element still exists. After a reload behind the dialog the Revoke button is
   200	  // a new element, so it is looked up again here. (No button any more: the drawer keeps the focus.)
   201	  const revokeButtonRef = useRef(null);
   202	  const wasConfirming = useRef(false);
   203	  useEffect(() => {
   204	    if (wasConfirming.current && confirming === null) revokeButtonRef.current?.focus();
   205	    wasConfirming.current = confirming !== null;
   206	  }, [confirming]);
   207	
   208	  function handleRevoked(updated) {
   209	    setRevoked(updated);
   210	    setConfirming(null);
   211	    onChanged();
   212	  }
   213	
   214	  // The revocation may or may not have happened: ask the server instead of showing our old copy,
   215	  // which would otherwise keep saying "Active" (with a Revoke button) for a revoked certificate.
   216	  function handleOutcomeUnknown() {
   217	    setOutcomeOpen(true);
   218	    setUnverified(onScreen.current); // read through the ref: this runs long after it was rendered
   219	    detail.reload();
   220	    onChanged();
   221	  }
   222	
   223	  return (
   224	    <>
   225	    <Dialog title="Certificate details" variant="drawer" onClose={onClose}>
   226	      <div className="drawer-header">
   227	        <h2>Certificate</h2>
   228	        <button type="button" className="btn btn-small" onClick={onClose} aria-label="Close details">
   229	          Close ✕
   230	        </button>
   231	      </div>
   232	
   233	      <div className="drawer-body" aria-busy={verifying} data-testid="certificate-detail">
   234	        {!certificate && detail.status === 'loading' ? (
   235	          <div className="state state-loading">
   236	            <Spinner label="Loading certificate…" />
   237	          </div>
   238	        ) : !certificate ? (
   239	          detail.error?.status === 404 ? (
   240	            <ErrorState
   241	              title="Certificate not found"
   242	              error={{ message: 'It does not exist in this tenant. It may belong to another tenant or the link may be outdated.' }}
   243	            />
   244	          ) : (
   245	            <ErrorState title="The certificate could not be loaded" error={detail.error} onRetry={detail.reload} />
   246	          )
   247	        ) : (
   248	          <>
   249	            {revoked ? (
   250	              <p className="note note-success" role="status" data-testid="revoke-result">
   251	                {revoked.alreadyRevoked
   252	                  ? 'This certificate had already been revoked; nothing was changed.'
   253	                  : 'The certificate has been revoked.'}
   254	              </p>
   255	            ) : null}
   256	
   257	            <div className="detail-title">
   258	              <span className="mono detail-serial" data-testid="detail-serial">
   259	                {certificate.serialNumber}
   260	              </span>
   261	              <span data-testid="detail-status">
   262	                {verifying ? <Spinner label="Checking the current status…" /> : <StatusBadge status={certificate.status} />}
   263	              </span>
   264	            </div>
   265	
   266	            <dl className="detail-grid">
   267	              <Field label="Learner">{certificate.learnerName}</Field>
   268	              <Field label="Course">{certificate.courseTitle}</Field>
   269	              <Field label={`Issue date (${certificate.tenant.timezone})`}>
   270	                <span data-testid="detail-issued-local">{certificate.issuedLocal}</span>
   271	                <span className="muted small"> UTC{certificate.issuedUtcOffset}</span>
   272	                <div className="muted small">UTC: {certificate.issuedAt}</div>
   273	              </Field>
   274	              <Field label="Issued by">
   275	                {certificate.issuedBy ?? '—'}
   276	                {certificate.bulkOperationId ? <span className="muted small"> · bulk operation #{certificate.bulkOperationId}</span> : null}
   277	              </Field>
   278	              {certificate.status === 'revoked' ? (
   279	                <>
   280	                  <Field label={`Revoked (${certificate.tenant.timezone})`}>
   281	                    {certificate.revokedLocal}
   282	                    <div className="muted small">UTC: {certificate.revokedAt}</div>
   283	                  </Field>
   284	                  <Field label="Revoked by">{certificate.revokedBy ?? '—'}</Field>
   285	                  <Field label="Revocation reason">{certificate.revocationReason ?? 'No reason given'}</Field>
   286	                </>
   287	              ) : null}
   288	            </dl>
   289	
   290	            {certificate.status === 'active' && !verifying ? (
   291	              <div className="detail-actions">
   292	                <button
   293	                  type="button"
   294	                  className="btn btn-danger"
   295	                  onClick={() => setConfirming(certificate)}
   296	                  ref={revokeButtonRef}
   297	                  data-testid="revoke-button"
   298	                >
   299	                  Revoke certificate…
   300	                </button>
   301	              </div>
   302	            ) : null}
   303	
   304	            <h3>Audit trail</h3>
   305	            <ol className="audit-list" data-testid="audit-trail">
   306	              {certificate.auditEvents.map((event) => (
   307	                <li key={event.id}>
   308	                  <div>
   309	                    <strong>{ACTION_LABELS[event.action] ?? event.action}</strong>
   310	                    <span className="muted"> by {event.actorEmail}</span>
   311	                  </div>
   312	                  <div className="muted small">
   313	                    {event.createdLocal} ({certificate.tenant.timezone})
   314	                    {event.details?.reason ? ` · reason: ${event.details.reason}` : ''}
   315	                    {event.bulkOperationId ? ` · bulk operation #${event.bulkOperationId}` : ''}
   316	                  </div>
   317	                </li>
   318	              ))}
   319	            </ol>
   320	          </>
   321	        )}
   322	      </div>
   323	    </Dialog>
   324	
   325	      {confirming !== null ? (
   326	        <RevokeDialog
   327	          api={api}
   328	          certificate={confirming}
   329	          outcomeOpen={outcomeOpen}
   330	          revokedOnServer={certificate?.status === 'revoked'}
   331	          onCancel={() => setConfirming(null)}
   332	          onRevoked={handleRevoked}
   333	          onOutcomeUnknown={handleOutcomeUnknown}
   334	        />
   335	      ) : null}
   336	    </>
   337	  );
   338	}
