    11	const ACTOR_STORAGE_KEY = 'cert-admin.actor';
    12	
    13	function readStoredActor() {
    14	  try {
    15	    return window.localStorage.getItem(ACTOR_STORAGE_KEY);
    16	  } catch {
    17	    return null;
    18	  }
    19	}
    20	
    21	function storeActor(email) {
    22	  try {
    23	    window.localStorage.setItem(ACTOR_STORAGE_KEY, email);
    24	  } catch {
    25	    // storage unavailable (private mode): the choice simply does not survive a reload
    26	  }
    27	}
    28	
    29	function Console() {
    30	  const actors = useQuery((signal) => fetchActors(signal), []);
    31	  const [selectedEmail, setSelectedEmail] = useState(readStoredActor);
    32	  const route = useHashRoute();
    33	  // Bulk-issue attempts outlive the Bulk issue page: leaving the tab while a request is pending
    34	  // must not drop its lock, its status or its idempotency key. The store also brings back, from
    35	  // the tab's sessionStorage, attempts that a reload interrupted (see bulkIssueStore.js).
    36	  const [bulkIssues] = useState(() => createBulkIssueStore());
    37	
    38	  const actor = useMemo(() => {
    39	    if (!actors.data?.length) return null;
    40	    return actors.data.find((candidate) => candidate.email === selectedEmail) ?? actors.data[0];
    41	  }, [actors.data, selectedEmail]);
    42	
    43	  const api = useMemo(() => (actor ? createApi(actor.email) : null), [actor]);
    44	
    45	  useEffect(() => {
    46	    document.title = actor ? `Certificates · ${actor.tenant.name}` : 'Certificate Admin Console';
    47	  }, [actor]);
    48	
    49	  function switchActor(email) {
    50	    if (email === actor?.email) return;
    51	    storeActor(email);
    52	    setSelectedEmail(email);
    53	    // Ids are tenant-specific: never carry an open certificate over to another tenant.
    54	    navigate(route.view === 'bulk-issue' ? '/bulk-issue' : '/certificates');
    55	  }
    56	
    57	  if (!actor) {
    58	    return (
    59	      <div className="boot">
    60	        {actors.status === 'error' ? (
    61	          <ErrorState title="The console could not start" error={actors.error} onRetry={actors.reload} />
    62	        ) : actors.status === 'success' ? (
    63	          <ErrorState title="No admin accounts found" error={{ message: 'The database has no seeded admins. Run "npm run db:reset".' }} />
    64	        ) : (
    65	          <Spinner label="Loading console…" />
    66	        )}
    67	      </div>
    68	    );
    69	  }
    70	
    71	  return (
    72	    <div className="app">
    73	      <header className="topbar">
    74	        <div className="topbar-brand">
    75	          <span className="brand-mark" aria-hidden="true">
    76	            ✓
    77	          </span>
    78	          <span className="brand-name">Certificate Admin</span>
    79	        </div>
    80	
    81	        <nav className="tabs" aria-label="Sections">
    82	          <a href="#/certificates" className={route.view === 'certificates' ? 'tab tab-active' : 'tab'} aria-current={route.view === 'certificates' ? 'page' : undefined}>
    83	            Certificates
    84	          </a>
    85	          <a href="#/bulk-issue" className={route.view === 'bulk-issue' ? 'tab tab-active' : 'tab'} aria-current={route.view === 'bulk-issue' ? 'page' : undefined}>
    86	            Bulk issue
    87	          </a>
    88	        </nav>
    89	
    90	        <div className="topbar-actor">
    91	          <div className="tenant-chip" title="All data and actions are limited to this tenant">
    92	            <span className="tenant-name" data-testid="tenant-name">
    93	              {actor.tenant.name}
    94	            </span>
    95	            <span className="tenant-zone">{actor.tenant.timezone}</span>
    96	          </div>
    97	          <label className="actor-select">
    98	            <span>Acting as</span>
    99	            <select value={actor.email} onChange={(event) => switchActor(event.target.value)} data-testid="actor-select">
   100	              {actors.data.map((candidate) => (
   101	                <option key={candidate.email} value={candidate.email}>
   102	                  {candidate.email} ({candidate.tenant.name})
   103	                </option>
   104	              ))}
   105	            </select>
   106	          </label>
   107	        </div>
   108	      </header>
   109	
   110	      {/* Keyed by actor: switching tenants discards every piece of page state (search, selection,
   111	          viewed results) and cancels an export in flight. Only bulk-issue attempts are kept, per
   112	          actor, in the store above; the page of another actor never reads them. */}
   113	      <main className="content" key={actor.email}>
   114	        {route.view === 'bulk-issue' ? (
   115	          <BulkIssuePage api={api} actor={actor} bulkIssues={bulkIssues} />
   116	        ) : (
   117	          <CertificatesPage api={api} actor={actor} selectedCertificateId={route.certificateId} />
   118	        )}
   119	      </main>
   120	    </div>
   121	  );
   122	}

   160	/**
   161	 * The search text as it is sent to the server. Pasted text easily carries tabs or line breaks
   162	 * (two spreadsheet cells, a line of an e-mail) and the server rejects control characters with a
   163	 * 400, so every run of them - together with the spaces around it - becomes a single space.
   164	 */
   165	function toSearchQuery(text) {
   166	  return text.replace(/ *[\u0000-\u001f\u007f][\u0000-\u001f\u007f ]*/g, ' ').trim();
   167	}

   169	function formatBytes(bytes) {
   170	  if (bytes < 1024) return `${bytes} B`;
   171	  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`;
   172	  return `${(bytes / 1024 / 1024).toFixed(1)} MB`;
   173	}

   518	export function Pagination({ page, totalPages, total, pageSize, itemCount, disabled, onPageChange, noun = 'certificates' }) {
   519	  const first = total === 0 ? 0 : (page - 1) * pageSize + 1;
   520	  const last = total === 0 ? 0 : first + itemCount - 1;
   521	  const format = (n) => n.toLocaleString('en-US');
   522	  return (
   523	    <nav className="pagination" aria-label="Pagination">
   524	      <span className="muted" aria-live="polite">
   525	        {total === 0 ? `No ${noun}` : `${format(first)}–${format(last)} of ${format(total)} ${noun}`}
   526	      </span>
   527	      <div className="pagination-buttons">
   528	        <button type="button" className="btn btn-small" disabled={disabled || page <= 1} onClick={() => onPageChange(1)}>
   529	          « First
   530	        </button>
   531	        <button type="button" className="btn btn-small" disabled={disabled || page <= 1} onClick={() => onPageChange(page - 1)}>
   532	          ‹ Previous
   533	        </button>
   534	        <span className="pagination-page">
   535	          Page {format(page)} of {format(totalPages)}
   536	        </span>
   537	        <button
   538	          type="button"
   539	          className="btn btn-small"
   540	          disabled={disabled || page >= totalPages}
   541	          onClick={() => onPageChange(page + 1)}
   542	        >

   549	          onClick={() => onPageChange(totalPages)}

   558	/**
   559	 * Modal layer used for the certificate drawer and the revoke confirmation.
   560	 * Escape and a click on the backdrop close it (unless `locked`, e.g. while a request is running);
   561	 * focus moves into the dialog on open and returns to the previously focused element on close.
   562	 */
   563	export function Dialog({ title, onClose, locked = false, variant = 'modal', children, labelledBy, describedBy }) {
   564	  const panelRef = useRef(null);
   565	  const onCloseRef = useRef(onClose);
   566	  onCloseRef.current = onClose;
   567	  const lockedRef = useRef(locked);
   568	  lockedRef.current = locked;
   569	
   570	  useEffect(() => {
   571	    const panel = panelRef.current;
   572	    const previouslyFocused = document.activeElement;
   573	    panel?.focus();
   574	    const onKeyDown = (event) => {
   575	      if (event.key !== 'Escape' || lockedRef.current) return;
   576	      // Only the top-most dialog reacts, so Escape closes the confirmation before the drawer.
   577	      const dialogs = document.querySelectorAll('[role="dialog"]');
   578	      if (dialogs[dialogs.length - 1] === panelRef.current) {
   579	        event.stopPropagation();
   580	        onCloseRef.current();
   581	      }
   582	    };
   583	    document.addEventListener('keydown', onKeyDown);
   584	    return () => {
   585	      document.removeEventListener('keydown', onKeyDown);
   586	      if (previouslyFocused instanceof HTMLElement && document.contains(previouslyFocused)) {
   587	        previouslyFocused.focus();
   588	        return;
   589	      }
   590	      // The element that opened this dialog no longer exists (the Revoke button of a certificate
   591	      // that is revoked now). The focus would then drop to <body>, behind a dialog that is still
   592	      // open, and the next Tab would leave it: hand the focus to the top-most remaining dialog.
   593	      const remaining = [...document.querySelectorAll('[role="dialog"]')].filter((dialog) => dialog !== panel);
   594	      remaining[remaining.length - 1]?.focus();
   595	    };
   596	  }, []);
   597	
   598	  return (
   599	    <div
   600	      className={`overlay overlay-${variant}`}
   601	      onMouseDown={(event) => {
   602	        if (event.target === event.currentTarget && !locked) onClose();
   603	      }}
   604	    >
   605	      <div
   606	        ref={panelRef}
   607	        className={`dialog dialog-${variant}`}
   608	        role="dialog"
   609	        aria-modal="true"
   610	        aria-label={labelledBy ? undefined : title}
   611	        aria-labelledby={labelledBy}
   612	        aria-describedby={describedBy}
   613	        tabIndex={-1}
   614	      >
   615	        {children}
   616	      </div>
   617	    </div>
   618	  );
   619	}

   629	/**
   630	 * Last line of defence. Without a boundary React unmounts the whole tree when rendering throws,
   631	 * leaving a blank page with no explanation. Malformed server answers are already turned into the
   632	 * regular error states by api.js; this catches what nobody anticipated (a bug, or data that passed
   633	 * those checks and still cannot be displayed).
   634	 *
   635	 * Reloading helps when the problem was a passing one. It does not help when the open page itself is
   636	 * what cannot be displayed: a deep link to a record with unusable data fails again after every
   637	 * reload. From such a link the panel therefore also leads back to the certificate list, the place
   638	 * every other page is reached from.
   639	 */
   640	export default class ErrorBoundary extends Component {
   641	  state = { error: null };
   642	
   643	  static getDerivedStateFromError(error) {
   644	    return { error };
   645	  }
   646	
   647	  render() {
   648	    const { error } = this.state;
   649	    if (error === null) return this.props.children;
   650	    const route = parseHash(window.location.hash);
   651	    const onCertificateList = route.view === 'certificates' && route.certificateId === null;
   652	    return (
   653	      <div className="boot">
   654	        <div className="state state-error" role="alert" data-testid="app-crashed">
   655	          <div className="state-icon" aria-hidden="true">
   656	            !
   657	          </div>
   658	          <div className="state-body">
   659	            <strong>The console ran into an unexpected problem</strong>
   660	            <p>
   661	              This page could not be displayed. Reload the console to continue.
   662	              {onCertificateList ? null : ' If the problem comes back, go to the certificate list instead.'}
   663	            </p>
   664	            {/* What the runtime reported is for whoever looks into the problem, not for the admin:
   665	                in a production build it is a minified framework message. */}
   666	            <details className="muted small">
   667	              <summary>Technical details</summary>
   668	              {String(error?.message || error)}
   669	            </details>
   670	          </div>
   671	          <div className="state-actions">
   672	            <button type="button" className="btn" onClick={() => window.location.reload()}>
   673	              Reload the console
   674	            </button>
   675	            {onCertificateList ? null : (
   676	              <button type="button" className="btn" onClick={goToCertificateList}>
   677	                Go to the certificate list
   678	              </button>
   679	            )}
   680	          </div>
   681	        </div>
   682	      </div>
   683	    );
   684	  }
   685	}
