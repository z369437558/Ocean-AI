    30	function formatBytes(bytes) {
    31	  if (bytes < 1024) return `${bytes} B`;
    32	  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`;
    33	  return `${(bytes / 1024 / 1024).toFixed(1)} MB`;
    34	}

   327	/** Confirmation step for revoking: nothing is sent until the admin explicitly confirms. */
   328	function RevokeDialog({ api, certificate, onCancel, onRevoked }) {
   329	  const [reason, setReason] = useState('');
   330	  const [state, setState] = useState({ status: 'idle', error: null });
   331	  const submitting = useRef(false);
   332	
   333	  async function confirm() {
   334	    if (submitting.current) return; // ignore double clicks while the request is on its way
   335	    submitting.current = true;
   336	    setState({ status: 'submitting', error: null });
   337	    try {
   338	      onRevoked(await api.revokeCertificate(certificate.id, reason.trim() || undefined));
   339	    } catch (error) {
   340	      submitting.current = false;
   341	      setState({ status: 'error', error });
   342	    }
   343	  }
   344	
   345	  const busy = state.status === 'submitting';
   346	  return (
   347	    <Dialog onClose={onCancel} locked={busy} labelledBy="revoke-title">
   348	      <form
   349	        className="dialog-body"
   350	        onSubmit={(event) => {
   351	          event.preventDefault();
   352	          confirm();
   353	        }}
   354	        data-testid="revoke-dialog"
   355	      >
   356	        <h2 id="revoke-title">Revoke this certificate?</h2>
   357	        <p>
   358	          <span className="mono">{certificate.serialNumber}</span> was issued to <strong>{certificate.learnerName}</strong> for{' '}
   359	          <strong>{certificate.courseTitle}</strong>. After revocation it is no longer valid. This cannot be undone, and
   360	          the action is recorded in the audit trail.
   361	        </p>
   362	        <label className="field">
   363	          <span className="field-label">Reason (optional)</span>
   364	          <textarea
   365	            rows={3}
   366	            maxLength={MAX_REASON_LENGTH}
   367	            value={reason}
   368	            disabled={busy}
   369	            placeholder="e.g. Issued in error"
   370	            onChange={(event) => setReason(event.target.value)}
   371	            data-testid="revoke-reason"
   372	          />
   373	        </label>
   374	        {state.status === 'error' ? (
   375	          <p className="note note-error" role="alert">
   376	            The certificate was not revoked: {state.error.message}
   377	          </p>
   378	        ) : null}
   379	        <div className="dialog-actions">
   380	          <button type="button" className="btn" onClick={onCancel} disabled={busy}>
   381	            Cancel
   382	          </button>
   383	          <button type="submit" className="btn btn-danger" disabled={busy} data-testid="revoke-confirm">
   384	            {busy ? <Spinner label="Revoking…" /> : state.status === 'error' ? 'Try again' : 'Revoke certificate'}
   385	          </button>
   386	        </div>
   387	      </form>
   388	    </Dialog>
   389	  );
   390	}

   401	export default function CertificateDetail({ api, certificateId, onClose, onChanged }) {
   402	  const detail = useQuery((signal) => api.getCertificate(certificateId, signal), [api, certificateId]);
   403	  const [revoked, setRevoked] = useState(null); // server state returned by a successful revoke
   404	  const [confirming, setConfirming] = useState(false);
   405	  const certificate = revoked ?? detail.data;
   406	
   407	  function handleRevoked(updated) {
   408	    setRevoked(updated);
   409	    setConfirming(false);
   410	    onChanged();
   411	  }
   412	
   413	  return (
   414	    <>
   415	    <Dialog title="Certificate details" variant="drawer" onClose={onClose}>
   416	      <div className="drawer-header">
   417	        <h2>Certificate</h2>
   418	        <button type="button" className="btn btn-small" onClick={onClose} aria-label="Close details">
   419	          Close ✕
   420	        </button>
   421	      </div>
   422	
   423	      <div className="drawer-body" data-testid="certificate-detail">
   424	        {!certificate && detail.status === 'loading' ? (
   425	          <div className="state state-loading">
   426	            <Spinner label="Loading certificate…" />
   427	          </div>
   428	        ) : !certificate ? (
   429	          detail.error?.status === 404 ? (
   430	            <ErrorState
   431	              title="Certificate not found"
   432	              error={{ message: 'It does not exist in this tenant. It may belong to another tenant or the link may be outdated.' }}
   433	            />
   434	          ) : (
   435	            <ErrorState title="The certificate could not be loaded" error={detail.error} onRetry={detail.reload} />
   436	          )
   437	        ) : (
   438	          <>
   439	            {revoked ? (
   440	              <p className="note note-success" role="status" data-testid="revoke-result">
   441	                {revoked.alreadyRevoked
   442	                  ? 'This certificate had already been revoked; nothing was changed.'
   443	                  : 'The certificate has been revoked.'}
   444	              </p>
   445	            ) : null}
   446	
   447	            <div className="detail-title">
   448	              <span className="mono detail-serial" data-testid="detail-serial">
   449	                {certificate.serialNumber}
   450	              </span>
   451	              <span data-testid="detail-status">
   452	                <StatusBadge status={certificate.status} />
   453	              </span>
   454	            </div>
   455	
   456	            <dl className="detail-grid">
   457	              <Field label="Learner">{certificate.learnerName}</Field>
   458	              <Field label="Course">{certificate.courseTitle}</Field>
   459	              <Field label={`Issue date (${certificate.tenant.timezone})`}>
   460	                <span data-testid="detail-issued-local">{certificate.issuedLocal}</span>
   461	                <span className="muted small"> UTC{certificate.issuedUtcOffset}</span>
   462	                <div className="muted small">UTC: {certificate.issuedAt}</div>
   463	              </Field>
   464	              <Field label="Issued by">
   465	                {certificate.issuedBy ?? '—'}
   466	                {certificate.bulkOperationId ? <span className="muted small"> · bulk operation #{certificate.bulkOperationId}</span> : null}
   467	              </Field>
   468	              {certificate.status === 'revoked' ? (
   469	                <>
   470	                  <Field label={`Revoked (${certificate.tenant.timezone})`}>
   471	                    {certificate.revokedLocal}
   472	                    <div className="muted small">UTC: {certificate.revokedAt}</div>
   473	                  </Field>
   474	                  <Field label="Revoked by">{certificate.revokedBy ?? '—'}</Field>
   475	                  <Field label="Revocation reason">{certificate.revocationReason ?? 'No reason given'}</Field>
   476	                </>
   477	              ) : null}
   478	            </dl>
   479	
   480	            {certificate.status === 'active' ? (
   481	              <div className="detail-actions">
   482	                <button type="button" className="btn btn-danger" onClick={() => setConfirming(true)} data-testid="revoke-button">
   483	                  Revoke certificate…
   484	                </button>
   485	              </div>
   486	            ) : null}
   487	
   488	            <h3>Audit trail</h3>
   489	            <ol className="audit-list" data-testid="audit-trail">
   490	              {certificate.auditEvents.map((event) => (
   491	                <li key={event.id}>
   492	                  <div>
   493	                    <strong>{ACTION_LABELS[event.action] ?? event.action}</strong>
   494	                    <span className="muted"> by {event.actorEmail}</span>
   495	                  </div>
   496	                  <div className="muted small">
   497	                    {event.createdLocal} ({certificate.tenant.timezone})
   498	                    {event.details?.reason ? ` · reason: ${event.details.reason}` : ''}
   499	                    {event.bulkOperationId ? ` · bulk operation #${event.bulkOperationId}` : ''}
   500	                  </div>
   501	                </li>
   502	              ))}
   503	            </ol>
   504	          </>
   505	        )}
   506	      </div>
   507	    </Dialog>
   508	
   509	      {confirming && certificate ? (
   510	        <RevokeDialog api={api} certificate={certificate} onCancel={() => setConfirming(false)} onRevoked={handleRevoked} />
   511	      ) : null}
   512	    </>
   513	  );
   514	}

   538	/** Error panel with the server's message and, when it makes sense, a way to try again. */
   539	export function ErrorState({ title = 'Something went wrong', error, onRetry, retryLabel = 'Try again' }) {
   540	  return (
   541	    <div className="state state-error" role="alert">

   549	      </div>

   551	        <button type="button" className="btn" onClick={onRetry}>
   552	          {retryLabel}
   553	        </button>
   554	      ) : null}
   555	    </div>
   556	  );
   557	}

   571	export function Pagination({ page, totalPages, total, pageSize, itemCount, disabled, onPageChange, noun = 'certificates' }) {
   572	  const first = total === 0 ? 0 : (page - 1) * pageSize + 1;
   573	  const last = total === 0 ? 0 : first + itemCount - 1;
   574	  const format = (n) => n.toLocaleString('en-US');
   575	  return (
   576	    <nav className="pagination" aria-label="Pagination">
   577	      <span className="muted" aria-live="polite">
   578	        {total === 0 ? `No ${noun}` : `${format(first)}–${format(last)} of ${format(total)} ${noun}`}
   579	      </span>
   580	      <div className="pagination-buttons">
   581	        <button type="button" className="btn btn-small" disabled={disabled || page <= 1} onClick={() => onPageChange(1)}>
   582	          « First
   583	        </button>
   584	        <button type="button" className="btn btn-small" disabled={disabled || page <= 1} onClick={() => onPageChange(page - 1)}>
   585	          ‹ Previous
   586	        </button>
   587	        <span className="pagination-page">
   588	          Page {format(page)} of {format(totalPages)}
   589	        </span>
   590	        <button
   591	          type="button"
   592	          className="btn btn-small"
   593	          disabled={disabled || page >= totalPages}
   594	          onClick={() => onPageChange(page + 1)}
   595	        >
   596	          Next ›
   597	        </button>
   598	        <button
   599	          type="button"
   600	          className="btn btn-small"
   601	          disabled={disabled || page >= totalPages}
   602	          onClick={() => onPageChange(totalPages)}
   603	        >
   604	          Last »
   605	        </button>
   606	      </div>
   607	    </nav>
   608	  );
   609	}

   611	/**
   612	 * Modal layer used for the certificate drawer and the revoke confirmation.
   613	 * Escape and a click on the backdrop close it (unless `locked`, e.g. while a request is running);
   614	 * focus moves into the dialog on open and returns to the previously focused element on close.
   615	 */
   616	export function Dialog({ title, onClose, locked = false, variant = 'modal', children, labelledBy }) {
   617	  const panelRef = useRef(null);
   618	  const onCloseRef = useRef(onClose);
   619	  onCloseRef.current = onClose;
   620	  const lockedRef = useRef(locked);
   621	  lockedRef.current = locked;
   622	
   623	  useEffect(() => {
   624	    const previouslyFocused = document.activeElement;
   625	    panelRef.current?.focus();
   626	    const onKeyDown = (event) => {
   627	      if (event.key !== 'Escape' || lockedRef.current) return;
   628	      // Only the top-most dialog reacts, so Escape closes the confirmation before the drawer.
   629	      const dialogs = document.querySelectorAll('[role="dialog"]');
   630	      if (dialogs[dialogs.length - 1] === panelRef.current) {
   631	        event.stopPropagation();
   632	        onCloseRef.current();
   633	      }
   634	    };
   635	    document.addEventListener('keydown', onKeyDown);
   636	    return () => {
   637	      document.removeEventListener('keydown', onKeyDown);
   638	      if (previouslyFocused instanceof HTMLElement && document.contains(previouslyFocused)) previouslyFocused.focus();
   639	    };
   640	  }, []);
   641	
   642	  return (
   643	    <div
   644	      className={`overlay overlay-${variant}`}
   645	      onMouseDown={(event) => {
   646	        if (event.target === event.currentTarget && !locked) onClose();
   647	      }}
   648	    >
   649	      <div
   650	        ref={panelRef}
   651	        className={`dialog dialog-${variant}`}
   652	        role="dialog"
   653	        aria-modal="true"
   654	        aria-label={labelledBy ? undefined : title}
   655	        aria-labelledby={labelledBy}
   656	        tabIndex={-1}
   657	      >
   658	        {children}
   659	      </div>
   660	    </div>
   661	  );
   662	}
