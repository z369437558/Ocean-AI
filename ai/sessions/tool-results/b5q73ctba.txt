    11	/** One key per user-initiated operation; a retry of that operation re-sends the same key. */
    12	function newIdempotencyKey() {
    13	  if (typeof crypto.randomUUID === 'function') return crypto.randomUUID();
    14	  // crypto.randomUUID needs a secure context; getRandomValues works on plain http://host:port too.
    15	  const bytes = crypto.getRandomValues(new Uint8Array(16));
    16	  bytes[6] = (bytes[6] & 0x0f) | 0x40;
    17	  bytes[8] = (bytes[8] & 0x3f) | 0x80;
    18	  const hex = Array.from(bytes, (byte) => byte.toString(16).padStart(2, '0')).join('');
    19	  return `${hex.slice(0, 8)}-${hex.slice(8, 12)}-${hex.slice(12, 16)}-${hex.slice(16, 20)}-${hex.slice(20)}`;
    20	}

     5	/**
     6	 * Bulk-issue attempts, kept above the routed pages.
     7	 *
     8	 * The Bulk issue page is unmounted whenever the admin opens another tab or follows a serial link.
     9	 * If the attempt lived in that page, coming back while the request is still on its way would show
    10	 * an idle form and offer a fresh submission under a new idempotency key - a second operation for
    11	 * one intention. So the attempt (course + key), its status and the in-flight lock live here
    12	 * instead, one entry per actor e-mail:
    13	 *
    14	 * - a page that mounts again re-attaches to the attempt of its actor (lock, status, key);
    15	 * - a request that finishes while nobody is looking still records its result or its failure;
    16	 * - an actor only ever reads its own entry, so one tenant's operation is never shown while acting
    17	 *   as another tenant, and it is still there when the admin switches back.

    20	 * back) ends that lifetime while a request may still be on its way, so every attempt whose outcome
    21	 * is not known yet is also written to the tab's sessionStorage and picked up again by the next
    22	 * store - see `restore` below.

    41	  attempt !== null &&

    47	/**
    48	 * The attempts a previous page left without a known outcome, as store entries.
    49	 *
    50	 * Each one comes back as a failed run with an unknown (or "has not run yet") outcome, i.e. as the
    51	 * prompt to retry safely: the page that sent the request is gone and nobody will ever see its
    52	 * answer, but the key is still known, and repeating the request with that key is how to find out
    53	 * what happened without issuing twice.
    54	 */
    55	function restore(storage) {
    56	  const entries = new Map();
    57	  let saved;
    58	  try {
    59	    saved = JSON.parse(storage?.getItem(STORAGE_KEY) ?? 'null');
    60	  } catch {
    61	    return entries; // unreadable or not ours: start empty rather than fail to start at all
    62	  }
    63	  if (typeof saved !== 'object' || saved === null) return entries;
    64	  for (const [email, record] of Object.entries(saved)) {
    65	    if (!isAttempt(record?.attempt)) continue;
    66	    const { courseId, courseTitle, key } = record.attempt;
    67	    const failure = record.failure;
    68	    const error =
    69	      typeof failure === 'object' && failure !== null
    70	        ? new ApiError(Number(failure.status) || 0, String(failure.code), String(failure.message))
    71	        : new ApiError(0, 'interrupted', 'The page was reloaded or left before the answer from the server arrived.');
    72	    entries.set(email, { run: { status: 'failed', attempt: { courseId, courseTitle, key }, error }, settled: 0 });
    73	  }
    74	  return entries;
    75	}

    77	export function createBulkIssueStore(storage = tabStorage()) {
    78	  // actor e-mail -> { run, settled }
    79	  //   run:     { status: 'idle' }
    80	  //          | { status: 'submitting', attempt }
    81	  //          | { status: 'succeeded', attempt, operation, replayed }
    82	  //          | { status: 'failed', attempt, error }      with attempt = { courseId, courseTitle, key }
    83	  //   settled: number of requests of this actor that have finished; pages refetch what a bulk
    84	  //            issue changes (roster, history) whenever it moves.
    85	  const entries = restore(storage);
    86	  const listeners = new Set();
    87	
    88	  const read = (email) => entries.get(email) ?? NOTHING_YET;
    89	
    90	  function write(email, entry) {
    91	    entries.set(email, entry);
    92	    save();
    93	    for (const listener of listeners) listener();
    94	  }
    95	
    96	  /**
    97	   * Mirrors into the tab's storage what must outlive this page: the attempts that are on their way
    98	   * or wait for a retry, with their keys. A result, a rejection (4xx: nothing was written and the
    99	   * next attempt is a new operation) and a dismissed run are settled and are not kept.
   100	   */
   101	  function save() {
   102	    const unsettled = {};
   103	    for (const [email, { run }] of entries) {
   104	      if (run.status === 'submitting') {
   105	        unsettled[email] = { attempt: run.attempt };
   106	      } else if (run.status === 'failed' && writeOutcome(run.error) !== 'rejected') {
   107	        const { status = 0, code = 'unexpected_error', message = '' } = run.error ?? {};
   108	        unsettled[email] = { attempt: run.attempt, failure: { status, code, message: message || 'Unexpected error' } };
   109	      }
   110	    }
   111	    try {
   112	      if (Object.keys(unsettled).length > 0) storage?.setItem(STORAGE_KEY, JSON.stringify(unsettled));
   113	      else storage?.removeItem(STORAGE_KEY);
   114	    } catch {
   115	      // storage full or blocked: the attempt then lives as long as this page, as it did before
   116	    }
   117	  }
   118	
   119	  return {
   120	    read,
   121	
   122	    subscribe(listener) {
   123	      listeners.add(listener);
   124	      return () => listeners.delete(listener);
   125	    },
   126	
   127	    /**
   128	     * Sends `attempt` on behalf of the actor of `api`, unless that actor already has a request on
   129	     * its way. Checking and locking happen synchronously, before anything is rendered: a double
   130	     * click (or Enter + click) cannot start two requests, on whichever page instance it lands.
   131	     * Should a duplicate reach the server after all (e.g. a retry after a timeout), the
   132	     * idempotency key makes it harmless. Passing the attempt of a failed run again is the safe
   133	     * retry: same course, same key.
   134	     */
   135	    async submit(api, attempt) {
   136	      const email = api.actor;
   137	      const { run, settled } = read(email);
   138	      if (run.status === 'submitting') return;
   139	      write(email, { run: { status: 'submitting', attempt }, settled });
   140	
   141	      let next;
   142	      try {
   143	        const { operation, replayed } = await api.bulkIssue(attempt.courseId, attempt.key);
   144	        next = { status: 'succeeded', attempt, operation, replayed };
   145	      } catch (error) {
   146	        next = { status: 'failed', attempt, error };
   147	      }
   148	      // Nothing else can have touched the entry meanwhile: submit and dismiss both back off while
   149	      // a request is in flight.
   150	      write(email, { run: next, settled: settled + 1 });
   151	    },
   152	
   153	    /** Forgets a finished run (result or failure). A request in flight cannot be dismissed. */
   154	    dismiss(email) {
   155	      const { run, settled } = read(email);
   156	      if (run.status === 'submitting' || run.status === 'idle') return;
   157	      write(email, { run: IDLE, settled });
   158	    },
   159	  };
   160	}

   164	  return useSyncExternalStore(store.subscribe, () => store.read(email));

    97	/**
    98	 * Modal layer used for the certificate drawer and the revoke confirmation.
    99	 * Escape and a click on the backdrop close it (unless `locked`, e.g. while a request is running);
   100	 * focus moves into the dialog on open and returns to the previously focused element on close.
   101	 */
   102	export function Dialog({ title, onClose, locked = false, variant = 'modal', children, labelledBy, describedBy }) {
   103	  const panelRef = useRef(null);
   104	  const onCloseRef = useRef(onClose);
   105	  onCloseRef.current = onClose;
   106	  const lockedRef = useRef(locked);
   107	  lockedRef.current = locked;
   108	
   109	  useEffect(() => {
   110	    const panel = panelRef.current;
   111	    const previouslyFocused = document.activeElement;
   112	    panel?.focus();
   113	    const onKeyDown = (event) => {
   114	      if (event.key !== 'Escape' || lockedRef.current) return;
   115	      // Only the top-most dialog reacts, so Escape closes the confirmation before the drawer.
   116	      const dialogs = document.querySelectorAll('[role="dialog"]');
   117	      if (dialogs[dialogs.length - 1] === panelRef.current) {
   118	        event.stopPropagation();
   119	        onCloseRef.current();
   120	      }
   121	    };
   122	    document.addEventListener('keydown', onKeyDown);
   123	    return () => {
   124	      document.removeEventListener('keydown', onKeyDown);
   125	      if (previouslyFocused instanceof HTMLElement && document.contains(previouslyFocused)) {
   126	        previouslyFocused.focus();
   127	        return;
   128	      }
   129	      // The element that opened this dialog no longer exists (the Revoke button of a certificate
   130	      // that is revoked now). The focus would then drop to <body>, behind a dialog that is still
   131	      // open, and the next Tab would leave it: hand the focus to the top-most remaining dialog.
   132	      const remaining = [...document.querySelectorAll('[role="dialog"]')].filter((dialog) => dialog !== panel);
   133	      remaining[remaining.length - 1]?.focus();
   134	    };
   135	  }, []);
   136	
   137	  return (
   138	    <div
   139	      className={`overlay overlay-${variant}`}
   140	      onMouseDown={(event) => {
   141	        if (event.target === event.currentTarget && !locked) onClose();
   142	      }}
   143	    >
   144	      <div
   145	        ref={panelRef}
   146	        className={`dialog dialog-${variant}`}
   147	        role="dialog"
   148	        aria-modal="true"
   149	        aria-label={labelledBy ? undefined : title}
   150	        aria-labelledby={labelledBy}
   151	        aria-describedby={describedBy}
   152	        tabIndex={-1}
   153	      >
   154	        {children}
   155	      </div>
   156	    </div>
   157	  );
   158	}
